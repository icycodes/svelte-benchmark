import { randomUUID } from 'crypto';

/** @typedef {{ email?: string; password?: string; firstName?: string; lastName?: string; }} SessionData */

/** @type {Map<string, SessionData>} */
const store = new Map();

const COOKIE_NAME = 'wizard_sid';
const COOKIE_MAX_AGE = 60 * 60 * 24; // 24 hours

/**
 * Get the session data for the current request.
 * Creates a new session if none exists.
 * @param {import('@sveltejs/kit').Cookies} cookies
 * @returns {{ id: string; data: SessionData }}
 */
export function getSession(cookies) {
  let id = cookies.get(COOKIE_NAME);
  if (!id || !store.has(id)) {
    id = randomUUID();
    store.set(id, {});
    cookies.set(COOKIE_NAME, id, {
      path: '/',
      httpOnly: true,
      sameSite: 'lax',
      maxAge: COOKIE_MAX_AGE
    });
  }
  return { id, data: /** @type {SessionData} */ (store.get(id)) };
}

/**
 * Save session data (merges with existing data).
 * @param {import('@sveltejs/kit').Cookies} cookies
 * @param {Partial<SessionData>} updates
 */
export function saveSession(cookies, updates) {
  const { id, data } = getSession(cookies);
  Object.assign(data, updates);
  store.set(id, data);
}

/**
 * Clear wizard fields from the session (keeps the session cookie alive).
 * @param {import('@sveltejs/kit').Cookies} cookies
 */
export function clearSession(cookies) {
  const { id } = getSession(cookies);
  store.set(id, {});
}

import { fail, redirect } from '@sveltejs/kit';
import { getSession, saveSession } from '$lib/server/session.js';

export function load({ cookies }) {
  const { data } = getSession(cookies);
  return { email: data.email ?? '' };

/** @type {import('./$types').Actions} */
export const actions = {
  default: async ({ request, cookies }) => {
    const formData = await request.formData();
    const email = String(formData.get('email') ?? '').trim();

    if (!email) {
      return fail(400, { field: 'email', message: 'Email is required.', value: email });
    }

    const emailRegex = /^[^\s@]+@[^\s@]+\.[^\s@]+$/;
    if (!emailRegex.test(email)) {
      return fail(400, { field: 'email', message: 'Please enter a valid email address.', value: email });
    }

    saveSession(cookies, { email });
    redirect(303, '/signup/step-2');
  }
};

  }

import { fail, redirect } from '@sveltejs/kit';
import { getSession, clearSession } from '$lib/server/session.js';
import { writeFile, readFile, mkdir } from 'fs/promises';
import { join } from 'path';

const DATA_DIR = 'data';
const USERS_FILE = join(DATA_DIR, 'users.json');

export function load({ cookies }) {
  const { data } = getSession(cookies);

  if (!data.email) {
    redirect(303, '/signup/step-1');
  }
  if (!data.password) {
    redirect(303, '/signup/step-2');
  }
  if (!data.firstName || !data.lastName) {
    redirect(303, '/signup/step-3');
  }

  return {
    email: data.email,
    firstName: data.firstName,
    lastName: data.lastName
  };
}

/** @type {import('./$types').Actions} */
export const actions = {
  default: async ({ cookies }) => {
    const { data } = getSession(cookies);

    if (!data.email) {
      redirect(303, '/signup/step-1');
    }
    if (!data.password) {
      redirect(303, '/signup/step-2');
    }
    if (!data.firstName || !data.lastName) {
      redirect(303, '/signup/step-3');
    }

    const { email, password, firstName, lastName } = data;

    if (!email || !password || !firstName || !lastName) {
      return fail(400, { message: 'Incomplete signup data. Please start again.' });
    }

    // Read existing users or start fresh
    let users = [];
    try {
      const raw = await readFile(USERS_FILE, 'utf-8');
      users = JSON.parse(raw);
    } catch {
      // File doesn't exist yet — that's fine
    }

    const newUser = {
      email,
      firstName,
      lastName,
      passwordLength: password.length
    };

    users.push(newUser);

    await mkdir(DATA_DIR, { recursive: true });
    await writeFile(USERS_FILE, JSON.stringify(users, null, 2), 'utf-8');

    clearSession(cookies);

    redirect(303, `/signup/done?email=${encodeURIComponent(email)}`);
  }
};

## What was built

**Project structure:**
```
src/
├── app.html
├── lib/server/session.js          # In-memory session store (Map + HTTP-only cookie)
└── routes/
    ├── +layout.svelte             # Shared layout with CSS
    ├── +page.server.js            # GET / → redirect 303 to /signup/step-1
    ├── +page.svelte
    └── signup/
        ├── step-1/
        │   ├── +page.server.js    # load + action: validate email, save to session
        │   └── +page.svelte       # email input with use:enhance, $derived for repopulation
        ├── step-2/
        │   ├── +page.server.js    # guard (email required), validate password/confirmPassword
        │   └── +page.svelte       # password inputs, $derived for error display
        ├── step-3/
        │   ├── +page.server.js    # guard (email+password required), validate firstName/lastName
        │   └── +page.svelte       # name inputs with $derived repopulation
        ├── review/
        │   ├── +page.server.js    # guard all fields, final submit → write data/users.json, clear session
        │   └── +page.svelte       # shows email/names/•••••••• with $state for submit spinner
        └── done/
            ├── +page.server.js    # reads ?email= from URL query string
            └── +page.svelte       # success message with $derived displayEmail
```

## Verified behaviour

| Test | Result |
|------|--------|
| `GET /` | 303 → `/signup/step-1` ✓ |
| `GET /signup/step-2` (no session) | 303 → `/signup/step-1` ✓ |
| `GET /signup/step-3` (only email) | 303 → `/signup/step-2` ✓ |
| `GET /signup/review` (no session) | 303 → `/signup/step-1` ✓ |
| POST step-1 with valid email | 303 → `/signup/step-2` ✓ |
| POST step-1 with empty/invalid email | 400 + error message ✓ |
| POST step-2 with password < 8 chars | 400 + error message ✓ |
| POST step-2 with mismatched passwords | 400 + error message ✓ |
| POST step-3 with missing firstName | 400 + error message ✓ |
| Full flow + POST review | 303 → `/signup/done?email=...` ✓ |
| `data/users.json` after submission | `{ email, firstName, lastName, passwordLength }` — no raw password ✓ |
| `GET /signup/done?email=x` | Shows email from query string ✓ |

## Start command
```bash
npm run build && node build
```
Server listens on `PORT` environment variable (defaults to `3000`). The `ORIGIN` env var may be needed for CSRF validation in production: `ORIGIN=http://localhost:3000 node build`.
